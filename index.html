// solution_pwntools.py — recommended approach
from pwn import *

HOST = "challenge.flagvault.local"
PORT = 5555

io = remote(HOST, PORT)

# Read the welcome banner
io.recvuntil(b"Ready? GO!\n")

for i in range(100):
    # Receive the problem line: "Round X/100: A op B = ?"
    line = io.recvline().decode().strip()
    print(line)

    # Extract the expression (between ": " and " = ?")
    expr = line.split(": ")[1].replace(" = ?", "")

    # Evaluate it (safe: only +, -, *, // with integers)
    answer = eval(expr)
    print(f"  → {answer}")

    io.sendline(str(answer).encode())

    # Read the "Correct!" confirmation
    feedback = io.recvline().decode().strip()
    print(feedback)

# All 100 done — read the flag
flag_line = io.recvall().decode()
print(flag_line)

// solution_socket.py — no extra dependencies
import socket

HOST = "challenge.flagvault.local"
PORT = 5555

def recv_line(s):
    data = b""
    while not data.endswith(b"\n"):
        chunk = s.recv(1)
        if not chunk:
            break
        data += chunk
    return data.decode().strip()

with socket.socket() as s:
    s.connect((HOST, PORT))
    s.settimeout(10)

    # Drain the banner
    banner = recv_line(s)
    while "Ready? GO!" not in banner:
        print(banner)
        banner = recv_line(s)
    print(banner)

    for _ in range(100):
        line = recv_line(s)
        print(line)
        expr = line.split(": ")[1].replace(" = ?", "")
        ans  = eval(expr)
        s.sendall((str(ans) + "\n").encode())
        print(recv_line(s))

    # Receive flag
    try:
        while True:
            print(recv_line(s))
    except:
        pass

Background

The server presents 100 arithmetic problems. Each one must be answered within 3 seconds. At ~1 problem per second you'd need to type 100 answers in under 5 minutes — impossible by hand for random problems.


The solution is to write a script that connects over TCP, reads each problem, evaluates the expression in code, and sends the answer automatically. This is the core skill behind interactive CTF pwnables.
